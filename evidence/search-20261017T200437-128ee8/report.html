from bisect import bisect_left, bisect_right

def count_pairs(a: list[int], b: list[int], lower: int, upper: int) -> int:
    if lower > upper or not a or not b:
        return 0
    b_sorted = sorted(b)
    count = 0
    for x in a:
        low = lower - x
        high = upper - x
        left = bisect_left(b_sorted, low)
        right = bisect_right(b_sorted, high)
        count += right - left
    return count

from bisect import bisect_left, bisect_right

def count_pairs(a: list[int], b: list[int], lower: int, upper: int) -> int:
    if lower > upper or not a or not b:
        return 0
    # Sort the smaller list to minimize O(N log M) work
    if len(a) <= len(b):
        xs, ys = a, sorted(b)
    else:
        xs, ys = b, sorted(a)
    count = 0
    # Local bindings to avoid repeated global lookups
    b_left = bisect_left
    b_right = bisect_right
    for x in xs:
        low = lower - x
        high = upper - x
        left = b_left(ys, low)
        right = b_right(ys, high)
        count += right - left
    return count
