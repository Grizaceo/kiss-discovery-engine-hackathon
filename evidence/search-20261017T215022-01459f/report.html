from bisect import bisect_left, bisect_right

def count_pairs(a: list[int], b: list[int], lower: int, upper: int) -> int:
    if lower > upper:
        return 0
    # Sort a copy of b to avoid mutating the original
    sorted_b = sorted(b)
    count = 0
    for x in a:
        # Find range in sorted_b where lower <= x + y <= upper
        # This implies y >= lower - x and y <= upper - x
        low_bound = lower - x
        high_bound = upper - x
        left_idx = bisect_left(sorted_b, low_bound)
        right_idx = bisect_right(sorted_b, high_bound)
        count += right_idx - left_idx
    return count

Model explanation
We use a two-pointer approach on sorted copies of both lists. A helper function count_at_most(K) counts pairs with sum <= K in O(n+m) time by iterating through the first list and maintaining a pointer in the second. The result is count_at_most(upper) - count_at_most(lower - 1), achieving O(n log n + m log m) total complexity.
Code · actual provider response
def count_pairs(a: list[int], b: list[int], lower: int, upper: int) -> int:
    if lower > upper or not a or not b:
        return 0

    def count_at_most(K: int) -> int:
        A = sorted(a)
        B = sorted(b)
        count = 0
        j = len(B) - 1
        for x in A:
            while j >= 0 and x + B[j] > K:
                j -= 1
            if j < 0:
                break
            count += j + 1
        return count

    return count_at_most(upper) - count_at_most(lower - 1)

from bisect import bisect_left, bisect_right

def count_pairs(a: list[int], b: list[int], lower: int, upper: int) -> int:
    if lower > upper or not a or not b:
        return 0
    sorted_b = sorted(b)
    count = 0
    for x in a:
        low_bound = lower - x
        high_bound = upper - x
        left = bisect_left(sorted_b, low_bound)
        right = bisect_right(sorted_b, high_bound)
        count += right - left
    return count
